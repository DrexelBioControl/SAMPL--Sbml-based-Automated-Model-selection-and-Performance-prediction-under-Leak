Rate Equations

uRSDg' = k_txn*RSD_temp - kRz*uRSDg
RSDg' = kRz*uRSDg - ksd*RSDg*IN + krev*I_RSDg*OUT - klkg*RSDg*F
IN' = k_txn*IN_temp - ksd*RSDg*IN + krev*I_RSDg*OUT + kfsd*I_RSDg*F
OUT' = ksd*RSDg*IN - kf_rep*OUT*DRL - krev*I_RSDg*OUT + klkg*RSDg*F + basal_frac*k_txn*RSD_temp
DRL' = -kf_rep*OUT*DRL
ROL' = kf_rep*OUT*DRL
I_RSDg' = ksd*RSDg*IN - krev*I_RSDg*OUT - kfsd*I_RSDg*F
F' = k_txn*F_temp - kfsd*I_RSDg*F + ksd*F_RSDg*IN - klkg*RSDg*F
F_RSDg' = kfsd*I_RSDg*F - ksd*F_RSDg*IN + klkg*RSDg*F
